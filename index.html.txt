  <script>
    const facts = [
      "Muraflex systems can reach standard heights of up to 10 feet without requiring a clerestory.",
      "The FINO system supports curved glass applications, creating flow and continuity with a minimum radius of 12 feet.",
      "FINO is designed with an ultra-thin profile, delivering a clean, continuous vision line and a minimal storefront look.",
      "The REVEAL system requires a minimum bulkhead ceiling clearance of 5 inches for proper installation.",
      "REVEAL incorporates a recessed profile that enhances depth and creates a continuous, floating architectural line.",
      "The Click System, available in REVEAL and FINO, fully seals sliding doors to enhance acoustic performance.",
      "EXPO's motorized walls conceal the motor within the upper track for a clean and streamlined design.",
      "QUADRO is known for its framed visual effect, featuring snap-on contrasting frames that create a bold architectural statement.",
      "PLANIKA integrates seamlessly with furniture systems to create fully functional, adaptable workspaces.",
      "The anodization process uses an acid bath and electrical current to enhance aluminum durability while influencing its final finish.",
      "The powder coating process allows Muraflex systems to achieve virtually any color using RAL specifications.",
      "Muraflex systems are Greenguard Gold Certified and can contribute to LEED and other sustainability initiatives."
    ];

    const blockIndices = [0, 1, 2];
    const blockTimers = [null, null, null];
    let factIndices = [0, 1, 2];

    function initializeBlocks() {
      blockIndices.forEach((blockId, idx) => {
        const block = document.getElementById(`block-${blockId}`);
        const factDiv = block.querySelector('.fact');
        factDiv.innerHTML = `<div class="fact-content">${facts[factIndices[idx]]}</div>`;
        factDiv.classList.add('active');
      });

      updateIndicators();
    }

    function rotateBlock(blockIdx) {
      const block = document.getElementById(`block-${blockIdx}`);
      const factDiv = block.querySelector('.fact');
      
      factDiv.classList.remove('active');
      factDiv.classList.add('fade-out');

      setTimeout(() => {
        factIndices[blockIdx] = (factIndices[blockIdx] + 1) % facts.length;
        factDiv.innerHTML = `<div class="fact-content">${facts[factIndices[blockIdx]]}</div>`;
        factDiv.classList.remove('fade-out');
        factDiv.classList.add('active');
        updateIndicators();
      }, 800);
    }

    function updateIndicators() {
      const indicatorDiv = document.getElementById('indicator');
      indicatorDiv.innerHTML = '';
      
      facts.forEach((_, idx) => {
        const dot = document.createElement('div');
        dot.className = 'dot';
        if (factIndices.includes(idx)) {
          dot.classList.add('active');
        }
        dot.addEventListener('click', () => {
          const blockIdx = factIndices.indexOf(idx);
          if (blockIdx !== -1) {
            factIndices[blockIdx] = idx;
            const block = document.getElementById(`block-${blockIdx}`);
            const factDiv = block.querySelector('.fact');
            factDiv.innerHTML = `<div class="fact-content">${facts[idx]}</div>`;
            updateIndicators();
          }
        });
        indicatorDiv.appendChild(dot);
      });
    }

    function startRotation() {
      blockIndices.forEach((_, blockIdx) => {
        blockTimers[blockIdx] = setInterval(() => {
          rotateBlock(blockIdx);
        }, 10000);
      });
    }

    initializeBlocks();
    startRotation();
  </script>
